import java.io.*;
import java.util.*;

class Result {

    public static void matrixRotation(List<List<Integer>> matrix, int r) {

        int m = matrix.size();
        int n = matrix.get(0).size();

        int[][] a = new int[m][n];

        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                a[i][j] = matrix.get(i).get(j);
            }
        }

        int layers = Math.min(m, n) / 2;

        for (int layer = 0; layer < layers; layer++) {

            int top = layer;
            int left = layer;
            int bottom = m - 1 - layer;
            int right = n - 1 - layer;

            List<Integer> values = new ArrayList<>();

            for (int i = top; i <= bottom; i++) {
                values.add(a[i][left]);
            }

            for (int j = left + 1; j <= right; j++) {
                values.add(a[bottom][j]);
            }

            for (int i = bottom - 1; i >= top; i--) {
                values.add(a[i][right]);
            }

            for (int j = right - 1; j > left; j--) {
                values.add(a[top][j]);
            }

            int size = values.size();

            int shift = r % size;

            int index = shift;

            for (int i = top; i <= bottom; i++) {
                a[i][left] = values.get(index);
                index = (index + 1) % size;
            }

            for (int j = left + 1; j <= right; j++) {
                a[bottom][j] = values.get(index);
                index = (index + 1) % size;
            }

            for (int i = bottom - 1; i >= top; i--) {
                a[i][right] = values.get(index);
                index = (index + 1) % size;
            }

            for (int j = right - 1; j > left; j--) {
                a[top][j] = values.get(index);
                index = (index + 1) % size;
            }
        }

        StringBuilder sb = new StringBuilder();

        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (j > 0) {
                    sb.append(" ");
                }
                sb.append(a[i][j]);
            }
            sb.append("\n");
        }

        System.out.print(sb.toString());
    }
}

public class Solution {

    public static void main(String[] args) throws IOException {

        BufferedReader bufferedReader =
                new BufferedReader(new InputStreamReader(System.in));

        StringTokenizer st =
                new StringTokenizer(bufferedReader.readLine());

        int m = Integer.parseInt(st.nextToken());
        int n = Integer.parseInt(st.nextToken());
        int r = Integer.parseInt(st.nextToken());

        List<List<Integer>> matrix = new ArrayList<>();

        for (int i = 0; i < m; i++) {

            st = new StringTokenizer(bufferedReader.readLine());

            List<Integer> row = new ArrayList<>();

            for (int j = 0; j < n; j++) {
                row.add(Integer.parseInt(st.nextToken()));
            }

            matrix.add(row);
        }

        Result.matrixRotation(matrix, r);

        bufferedReader.close();
    }
}
